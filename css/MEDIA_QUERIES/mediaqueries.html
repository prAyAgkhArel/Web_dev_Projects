Media Queries

Syntax:

Notice the change in color when extending the browser width

/*@media [type] [feature]*/


  • @media screen and (min-width: 768px) {}
  • @media print{}
  • @media(max-width: 600px) {}

/*Multiple conditions*/


  • @media (min-width: 768px) and (max-width: 1024px) {}
  • @media(min-width: 600px), (orientation:landscape) {}
  • @media not screen {}
  • @media not (min-width: 768px){}

/*Modern less than 640px -> mobile phones*/


  • @media(min-width: 640px ){} /*sm small tablets and up*/
  • @media(min-width: 768px ){} /*md tablets and up*/
  • @media(min-width: 1024px ){} /*lg laptops and up*/
  • @media(min-width: 1280px ){} /*xl desktops*/
  • @media(min-width: 1536px ){} /*2xl wide screens*/

  • @media(prefers-color-scheme: dark){
    :root{--bg: #0f172a; --text: #e2e8f0;; }
    } /*dark mode*/
  • @media(prefers-reduced-motion: reduce ){
    *{
    animation:none !important; transition:none !important; }
    } /*reduced motion*/
  • @media(forced-colors: active ){}
    @media(prefers-contrast: more){} /*High Contrast*/
  • @media(pointer: coarse){}
  • @media(pointer: fine){}
  • @media(hover: hover){}
  • @media(hover: none){}
  • @media <>nav, aside, .no-print{display:none;}
    body{font-size:12pt; color: black;} /*print*/
  • @media(orientation: portrait){}
    @media(orientation: landscape){}
